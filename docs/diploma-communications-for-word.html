    sender_id = Column(Integer, ForeignKey("users.id"))
    sender_role = Column(String(50), nullable=False)
    text = Column(String(2000), nullable=False)

def upsert_complaint_conversation(db, complaint):
    conversation = db.query(Conversation).filter(
        Conversation.type == "complaint",
        Conversation.complaint_id == complaint.id,
    ).first()
    if not conversation:
        conversation = Conversation(
            type="complaint", buyer_id=complaint.user_id,
            complaint_id=complaint.id, status="open",
        )
        db.add(conversation)
    db.commit()
    return conversation

@router.post("/admin/{complaint_id}/status")
def update_status(complaint_id, status, response_text, ...):
    complaint.status = status
    conversation = upsert_complaint_conversation(db, complaint)
    if response_text:
        complaint.admin_response = response_text[:2000]
        db.add(Notification(user_id=complaint.user_id, type="system",
            subject=f"Ответ на жалобу #{complaint.id}", body=...))
        db.add(Message(conversation_id=conversation.id,
            sender_id=user.id, text=response_text[:2000]))
    db.commit()

def send_notification(db, user_id, type, subject, body):
    db.add(Notification(user_id=user_id, type=type,
        subject=subject, body=body, is_read=0))
    db.commit()

@router.get("/unread-count")
def unread_count(request, db):
    count = db.query(Notification).filter(
        Notification.user_id == user.id,
        Notification.is_read == 0).count()
    return {"count": count}
